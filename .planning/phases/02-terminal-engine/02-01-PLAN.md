```go
// internal/session — public surface (Wave 2 consumes this; keep names exact)
type Status string
const (
    StatusRunning Status = "running"
    StatusExited  Status = "exited"
)

type Info struct {                // JSON-ready snapshot for REST
    ID        string    `json:"id"`
    Label     string    `json:"label"`
    Status    Status    `json:"status"`
    ExitCode  *int      `json:"exitCode,omitempty"`
    CreatedAt time.Time `json:"createdAt"`
}

type Manager struct{ ... }
func NewManager() *Manager
func (m *Manager) Spawn() (*Session, error)         // bash, cwd=home, label "bash #N"
func (m *Manager) Get(id string) (*Session, bool)
func (m *Manager) List() []Info                     // stable order: CreatedAt desc
func (m *Manager) Remove(id string) error           // only exited; ErrStillRunning otherwise

type Session struct{ ... }
func (s *Session) Info() Info
func (s *Session) Attach(connID string) <-chan []byte // replay queued first; buffered 256
func (s *Session) Detach(connID string)
func (s *Session) WriteInput(p []byte) error
func (s *Session) Resize(cols, rows uint16, forceRedraw bool) error // jiggle when equal+force
func (s *Session) Snapshot() []byte                  // opaque — Phase 4 swaps the ring behind this
func (s *Session) Stop()                             // D-14; idempotent; blocks until done
func (s *Session) Done() <-chan struct{}             // closed after exit watcher finishes
var ErrStillRunning = errors.New("session still running")
```

      ```go
      s.markExited(code) // sets status+exitCode under mu, notifies/closes conn queues
      _ = ptmx.Close()   // AFTER Wait
      close(s.done)
      ```

      ```go
      pgid := s.cmd.Process.Pid           // session leader ⇒ PGID == PID
      _ = syscall.Kill(-pgid, syscall.SIGTERM) // ESRCH ignored

      case <-s.done:
      ```
