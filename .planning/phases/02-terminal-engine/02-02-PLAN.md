// web/src/api/sessions.ts
export interface TermSession {
  id: string;          // opaque uuid — NEVER parse
  label: string;       // "bash #3" — server-assigned
  status: "running" | "exited";
  exitCode?: number;
  createdAt: string;   // RFC3339
}
export function useSessions(): UseQueryResult<TermSession[]>;      // queryKey ["sessions"], refetchInterval 5000
export function useSpawnSession(): UseMutationResult<TermSession>; // POST /api/sessions
export function useStopSession(): UseMutationResult<void, ApiError, string>;   // POST /api/sessions/{id}/stop
export function useDeleteSession(): UseMutationResult<void, ApiError, string>; // DELETE /api/sessions/{id}
