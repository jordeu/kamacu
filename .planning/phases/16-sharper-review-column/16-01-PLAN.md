       // fetchLists runs BOTH searches and applies top-precedence dedup (D-14).

       func fetchLists(ctx context.Context, repo, repoDir string) (PRLists, string, error) {
           awaiting, st, _ := listPRs(ctx, repo, repoDir, "user-review-requested:@me draft:false")
           if st != "ok" {

           }
           reviewed, st2, _ := listPRs(ctx, repo, repoDir, "reviewed-by:@me draft:false")
           if st2 != "ok" {
               return PRLists{}, st2, nil
           }
           reviewed = dedupeReviewed(awaiting, reviewed)
           return PRLists{Awaiting: awaiting, Reviewed: reviewed}, "ok", nil
       }

       // dedupeReviewed drops any reviewed PR whose Number appears in awaiting
       // (top section takes precedence, REVWD-03). Preserves reviewed order.
       func dedupeReviewed(awaiting, reviewed []PRSummary) []PRSummary {
           seen := make(map[int]struct{}, len(awaiting))
           for _, p := range awaiting {
               seen[p.Number] = struct{}{}
           }
           out := make([]PRSummary, 0, len(reviewed))
           for _, p := range reviewed {
               if _, dup := seen[p.Number]; dup {
                   continue
               }
               out = append(out, p)
           }
           return out
       }
