```go
package worktree

// Slug converts a task title to a ref-safe slug: lowercase, runs of chars
// outside [a-z0-9] collapse to a single '-', trim leading/trailing '-',
// cap at 40 chars (re-trim trailing '-'), empty result -> "task".
func Slug(title string) string

// Service serializes worktree mutations (coarse mutex — single-user app)
// and owns the central worktree root (D-23).
type Service struct { /* mu sync.Mutex; Root string */ }

// NewService returns a Service rooted at root (absolute, already ~-expanded,
// e.g. /home/user/.kangent/worktrees). Tests pass t.TempDir().
func NewService(root string) *Service

// PathFor returns Root/<filepath.Base(repoPath)>/<slug>-<taskID>.
func (s *Service) PathFor(repoPath, slug string, taskID int64) string

// ResolveBase returns a commit-ish to branch from per D-24. No network, ever.
func (s *Service) ResolveBase(ctx context.Context, repo string) (string, error)

// Create makes worktree at path on branch from base. Pre-checks the leaf
// path; reuses an existing branch when no worktree uses it (Pitfall 1).
func (s *Service) Create(ctx context.Context, repo, branch, path, base string) error

// DirtyCount returns the porcelain=v2 -uall record count for the worktree.
func (s *Service) DirtyCount(ctx context.Context, wt string) (int, error)

// Remove removes the worktree (force per caller), with the clean-but-
// submodules --force fallback, missing-path idempotency, and a trailing
// prune. NEVER touches branches (D-34).
func (s *Service) Remove(ctx context.Context, repo, wt string, force bool) error

// EnsureSubmodules best-effort initializes submodules if .gitmodules exists
// at the worktree root. Errors are returned for logging only — callers must
// not fail creation on them (D-25 spirit).
func (s *Service) EnsureSubmodules(ctx context.Context, wt string) error
```
