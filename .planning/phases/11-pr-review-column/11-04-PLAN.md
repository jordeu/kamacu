const { data: settings } = useSettings();

2. Collapse state — localStorage, per-project, default collapsed (supersedes D-05 per user request 2026-06-14; D-06/GHCOL-06). Key MUST be `kangent:review-collapsed:{projectId}`:
```tsx
const storageKey = `kangent:review-collapsed:${projectId}`;
const [collapsed, setCollapsed] = useState<boolean>(
  () => localStorage.getItem(storageKey) !== "0",  // absent => collapsed; only explicit "0" expands
);
const toggle = () => setCollapsed((c) => {
  const next = !c;
  localStorage.setItem(storageKey, next ? "1" : "0");
  return next;
});
```
