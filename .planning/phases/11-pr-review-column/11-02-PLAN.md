```go
// PullRequestRoutes registers GET /api/projects/{id}/pull-requests[?refresh=1].
// Always 200 — the Result.State field carries degradation (GHSET-03). The
// endpoint is the GHSET-02 backend enforcement point: it returns state=disabled
// when github_integration is off OR the project is unlinked, BEFORE ever
// touching gh.
func PullRequestRoutes(mux *http.ServeMux, db *sql.DB, svc *github.Service) {
    mux.HandleFunc("GET /api/projects/{id}/pull-requests", func(w http.ResponseWriter, r *http.Request) {
        id, ok := pathID(w, r)
        if !ok { return }

        // GATE 1 (GHSET-02): integration toggle off -> disabled, no gh.
        val, err := settings.Get(db, settings.KeyGithubIntegration)
        if err != nil {
            writeJSON(w, http.StatusOK, github.Result{State: "error"})
            return
        }
        if val != "on" {
            writeJSON(w, http.StatusOK, github.Result{State: "disabled"})
            return
        }

        // GATE 2: project not linked -> disabled, no gh. Also fetch repo_path
        // for cmd.Dir (Pitfall 6: gh host/account selection).
        var repo sql.NullString
        var repoPath string
        qerr := db.QueryRow(`SELECT github_repo, repo_path FROM projects WHERE id = ?`, id).Scan(&repo, &repoPath)
        if errors.Is(qerr, sql.ErrNoRows) {
            // Unknown project still returns disabled (200), NOT 404 — the
            // column never blocks on a project-lookup miss (degrade-don't-break).
            writeJSON(w, http.StatusOK, github.Result{State: "disabled"})
            return
        }
        if qerr != nil {
            writeJSON(w, http.StatusOK, github.Result{State: "error"})
            return
        }
        if !repo.Valid || strings.TrimSpace(repo.String) == "" {
            writeJSON(w, http.StatusOK, github.Result{State: "disabled"})
            return
        }

        force := r.URL.Query().Get("refresh") == "1"
        writeJSON(w, http.StatusOK, svc.Get(r.Context(), repo.String, repoPath, force))
    })
}
```
   Imports: `database/sql`, `errors`, `net/http`, `strings`, `kangent/internal/github`, `kangent/internal/settings`.

   IMPORTANT decisions baked in (do not deviate):
   - The toggle read uses `settings.Get(db, settings.KeyGithubIntegration)` — the existing read-at-use path with code-default "on" (RESEARCH Open Q1). Compare `val != "on"` so any non-"on" value (including a future "off") disables.
   - `cmd.Dir`/repoDir = `repo_path` (Pitfall 6) — the SELECT MUST fetch both `github_repo` and `repo_path`.
   - Always HTTP 200 for any valid id; `state` carries everything. The only non-200 is the 400 from `pathID` on a non-numeric id.
