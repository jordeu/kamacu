    ```ts
    export type Status = "todo" | "in_progress" | "in_review" | "done";
    export const STATUSES: Status[] = ["todo", "in_progress", "in_review", "done"];
    export const STATUS_LABELS: Record<Status, string> = {
      todo: "To Do", in_progress: "In Progress", in_review: "In Review", done: "Done",
    };
    export interface Project { id: number; name: string; repo_path: string; created_at: string; updated_at: string; }
    export interface Task { id: number; project_id: number; title: string; description: string; status: Status; position: number; created_at: string; updated_at: string; }
    ```

    `web/src/api/client.ts`:
    ```ts
    export class ApiError extends Error { constructor(message: string, public status: number) { super(message); } }
    export async function api<T>(path: string, init?: RequestInit): Promise<T>
    ```
    — fetch with `Content-Type: application/json` when a body is present; on non-2xx, parse body as `{error: string}` and throw `new ApiError(body.error ?? res.statusText, res.status)`; on 204 return `undefined as T`. Helpers `get/post/patch/del` wrapping it.

    `web/src/api/queries.ts` (query keys are the contract — feature plans invalidate these exact keys):
    - `useProjects()` → key `["projects"]`, GET /api/projects
    - `useTasks(projectId: number)` → key `["tasks", projectId]`, GET /api/projects/{id}/tasks, `enabled: !isNaN(projectId)`
    - `useTask(taskId: number)` → key `["task", taskId]`, GET /api/tasks/{id}

    `web/src/api/mutations.ts`:
    - `useCreateProject()` POST /api/projects `{name?, repo_path}` → invalidate `["projects"]`
    - `useRenameProject()` PATCH /api/projects/{id} `{name}` → invalidate `["projects"]`
    - `useDeleteProject()` DELETE → invalidate `["projects"]`
    - `useCreateTask(projectId)` POST /api/projects/{id}/tasks `{title, description?}` → invalidate `["tasks", projectId]`
    - `useUpdateTask(projectId)` PATCH /api/tasks/{id} `{title?, description?}` → invalidate `["tasks", projectId]` and `["task", id]`
    - `useDeleteTask(projectId)` DELETE /api/tasks/{id} → invalidate `["tasks", projectId]`
    - `useMoveTask(projectId)` POST /api/tasks/{id}/move `{status, after_id}` — full optimistic pattern from RESEARCH.md code example: `onMutate` cancelQueries + snapshot `["tasks", projectId]` + setQueryData applying the move locally (set status, resequence positions so the moved task sits after `after_id`/at top); `onError` restore snapshot; `onSettled` invalidateQueries. Export type `MoveArgs = { id: number; status: Status; afterId: number | null }`.

    `web/src/main.tsx`: StrictMode → `QueryClientProvider` (one `QueryClient` with `defaultOptions: { queries: { retry: 1, refetchOnWindowFocus: false } }`) → `BrowserRouter` → `<App />`. Import `./index.css`.
