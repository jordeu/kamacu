    ```go
    package store

    import (
        "database/sql"
        _ "modernc.org/sqlite" // driver name is "sqlite", NOT "sqlite3"
    )

    func Open(dbPath string) (*sql.DB, error) {
        dsn := "file:" + dbPath +
            "?_pragma=journal_mode(WAL)" +
            "&_pragma=busy_timeout(5000)" +
            "&_pragma=foreign_keys(1)" +
            "&_pragma=synchronous(NORMAL)" +
            "&_txlock=immediate"
        db, err := sql.Open("sqlite", dsn)
        if err != nil {
            return nil, err
        }
        db.SetMaxOpenConns(1) // single-writer discipline: eliminates SQLITE_BUSY
        return db, db.Ping()
    }
    ```

    Create `internal/store/migrate.go` — RESEARCH.md Pattern 3 verbatim: `//go:embed migrations/*.sql` into `migrationsFS embed.FS`; `Migrate(db *sql.DB) error` calls `goose.SetBaseFS(migrationsFS)`, `goose.SetDialect("sqlite3")` (dialect string is "sqlite3" even though driver name is "sqlite"), then `goose.Up(db, "migrations")`.
